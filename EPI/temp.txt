# ========================================================
# Matching diagnostics
# ========================================================

print()
print(
    "Matched class distribution"
)
print(
    "----------------------------------------"
)

matched_types = []
matched_distances = []

for anchor_idx, (
    gt_idx,
    distance,
) in matches.items():

    matched_types.append(
        gt.loc[
            gt_idx,
            "type"
        ]
    )

    matched_distances.append(
        distance
    )

print(
    pd.Series(
        matched_types
    ).value_counts()
)

if matched_distances:

    print()
    print(
        f"Match distance "
        f"min/mean/max : "
        f"{np.min(matched_distances):.3f} / "
        f"{np.mean(matched_distances):.3f} / "
        f"{np.max(matched_distances):.3f} sec"
    )


# ========================================================
# Unmatched diagnostics
# ========================================================

matched_anchor_indices = set(
    matches.keys()
)

matched_gt_indices = set(
    gt_idx
    for gt_idx, _
    in matches.values()
)

unmatched_anchors = (
    anchors[
        ~anchors.index.isin(
            matched_anchor_indices
        )
    ]
)

unmatched_gt = (
    gt[
        ~gt.index.isin(
            matched_gt_indices
        )
    ]
)

if len(unmatched_anchors):

    print()
    print(
        "Unmatched Anchors"
    )
    print(
        "----------------------------------------"
    )

    print(
        unmatched_anchors[
            [
                "event_id",
                "cst_peak_sec",
                "cst_peak_dy",
            ]
        ].to_string(
            index=False
        )
    )

if len(unmatched_gt):

    print()
    print(
        "Unmatched GT"
    )
    print(
        "----------------------------------------"
    )

    print(
        unmatched_gt[
            [
                "type",
                "slot",
                "start_sec",
                "end_sec",
            ]
        ].to_string(
            index=False
        )
    )
